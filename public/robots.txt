# Every group below is INDEPENDENT: a crawler obeys only the group that names
# it, and ignores the "*" group entirely. So the /install/ rule has to be
# repeated in each group — a single Disallow in the "*" group would leave the
# host-side setup wizard crawlable by every named bot here.

User-agent: *
Allow: /
Disallow: /install/
